"""Exact standalone vector receiver domain; created 2026-10-15 ET."""
from pathlib import Path
import pytest
from test_standard_range_model import _public_seed
from swdb import c_lowering

pytestmark = pytest.mark.skipif(bool(c_lowering.locate().missing), reason='Clang 22.1.8 is required')


def test_public_exact_index_body_reaches_wp_with_explicit_receiver_domain(tmp_path):
    _library, prove, marker = _public_seed(tmp_path)
    record = prove('--model', 'std_vector_int_operator_index')
    assert record['failure_class'] == 'parse_error', record
    assert marker.exists()  # parser-only stand-in, never a proof result
    cert = record['settings']['standard_vector_range_model']
    assert cert['acsl_input_premises_added'] == []
    assert cert['domain_role'] == 'explicit live typed vector representation; caller discharge required'
    assert cert['original_contract_sha256'] == cert['view_contract_sha256']
    assert cert['typed_input_domain']['extent_acsl'] == r'\at(this->capacity,Pre)'
    reports = {r['name']: r for r in record['evidence']['reports']}
    generated = Path(reports['wp_input']['path']).read_text()
    assert 'swdb_model_pointer_formation_' in generated
    assert r'\initialized(v->data +' in generated
    assert 'return this->data + n;' in generated
    assert record['settings']['verifier_launched'] is True
